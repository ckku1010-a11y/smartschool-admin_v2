// YCHLCP Room 207 ESP32 Electronic Doorplate & Check-In Button
#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>

const char* WIFI_SSID = "YCHLCP_WIFI";
const char* WIFI_PASS = "school_password";
String FIREBASE_ROOM_URL = "https://YOUR-PROJECT.firebasedatabase.app/ychlcp_slrs/rooms/207.json";
const int BTN_ARRIVE_PIN = 14; // Push button outside Room 207

void setup() {
  Serial.begin(115200);
  pinMode(BTN_ARRIVE_PIN, INPUT_PULLUP);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (WiFi.status() != WL_CONNECTED) delay(500);
}

void loop() {
  // 1. If cover teacher presses button at Room 207 door -> PATCH status="arrived"
  if (digitalRead(BTN_ARRIVE_PIN) == LOW) {
    HTTPClient http;
    http.begin(FIREBASE_ROOM_URL);
    http.addHeader("Content-Type", "application/json");
    http.sendRequest("PATCH", "{\"status\":\"arrived\",\"updatedBy\":\"ESP32_BTN_207\"}");
    http.end();
    delay(1000);
  }
  delay(200);
}
