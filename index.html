#include #include #include #include #include // ============================================================================ // 1. 請填寫你的手機熱點 Wi-Fi 及免費 Firebase 網址（結尾不要加斜線 /） // ============================================================================ const char* WIFI_SSID = "你的手機熱點名稱"; const char* WIFI_PASSWORD = "你的手機熱點密碼"; const String FIREBASE_BASE_URL = "https://你的專案名稱.asia-southeast1.firebasedatabase.app"; // ============================================================================ // 2. QAPASS 1602 I2C 設定 (常見位址為 0x27，部分型號為 0x3F) // ============================================================================ LiquidCrystal_I2C lcd(0x27, 16, 2); String currentLine1 = "RM207 SmartSLRS "; String currentLine2 = "WiFi Syncing... "; bool isLateAlert = false; bool backlightState = true; unsigned long lastPollTime = 0; unsigned long lastBlinkTime = 0; // 確保字串剛好 16 格，不會殘留舊字元 String format16(String text) { while (text.length() < 16) { text += " "; } return text.substring(0, 16); } // 更新 QAPASS 1602 螢幕上的兩行文字 void updateLcdText(String l1, String l2) { String clean1 = format16(l1); String clean2 = format16(l2); if (clean1 != currentLine1 || clean2 != currentLine2) { currentLine1 = clean1; currentLine2 = clean2; lcd.setCursor(0, 0); lcd.print(currentLine1); lcd.setCursor(0, 1); lcd.print(currentLine2); } } // 從 JSON 字串擷取指定欄位 String extractJsonString(const String& json, const String& key) { String pattern = "\"" + key + "\":\""; int idx = json.indexOf(pattern); if (idx == -1) return ""; int start = idx + pattern.length(); int end = json.indexOf("\"", start); if (end == -1) return ""; return json.substring(start, end); } // 向 Firebase 的 /lcd_door.json 讀取最新門外代課資訊 void fetchDoorLcdFromFirebase() { if (WiFi.status() != WL_CONNECTED) return; WiFiClientSecure client; client.setInsecure(); // 免憑證直接連線 HTTPS HTTPClient https; String url = FIREBASE_BASE_URL + "/lcd_door.json"; if (https.begin(client, url)) { int httpCode = https.GET(); if (httpCode == HTTP_CODE_OK) { String payload = https.getString(); if (payload != "null" && payload.length() > 10) { String l1 = extractJsonString(payload, "line1"); String l2 = extractJsonString(payload, "line2"); isLateAlert = (payload.indexOf("\"alert\":true") != -1); if (l1 != "" && l2 != "") { updateLcdText(l1, l2); } // 若無逾時警報，確保背光保持常亮 if (!isLateAlert && !backlightState) { backlightState = true; lcd.backlight(); } } } https.end(); } } void setup() { Serial.begin(115200); Wire.begin(21, 22); // SDA = GPIO 21, SCL = GPIO 22 lcd.init(); lcd.backlight(); updateLcdText("RM207 DOOR SIGN", "Connecting WiFi"); WiFi.begin(WIFI_SSID, WIFI_PASSWORD); int tries = 0; while (WiFi.status() != WL_CONNECTED && tries < 30) { delay(400); tries++; } if (WiFi.status() == WL_CONNECTED) { updateLcdText("WiFi Connected!", "Syncing Cloud..."); fetchDoorLcdFromFirebase(); } else { updateLcdText("R207 P4 3B CL_B1", "CKKU->TMCHAN ACK"); } } void loop() { // 1. 當網站觸發「防漏堂逾時提示」時，閃動 1602 背光燈吸引巡堂主任及老師注意 if (isLateAlert && (millis() - lastBlinkTime > 500)) { lastBlinkTime = millis(); backlightState = !backlightState; if (backlightState) lcd.backlight(); else lcd.noBacklight(); } // 2. 每 2 秒向 Firebase 同步一次門外代課狀態 if (millis() - lastPollTime > 2000) { lastPollTime = millis(); fetchDoorLcdFromFirebase(); } }
